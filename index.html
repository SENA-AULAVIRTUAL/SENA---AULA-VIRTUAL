-- ===================================================================== -- PLATAFORMA SENA - Horario por trimestre -- Pégalo completo en: Supabase > SQL Editor > New query > Run -- (Se ejecuta DESPUÉS de supabase-acceso.sql, una sola vez) -- ===================================================================== -- 1) TABLAS ------------------------------------------------------------ -- Un trimestre de una ficha (por ejemplo 2026-IV), con sus fechas create table if not exists trimestres ( ficha text not null references fichas(numero) on delete cascade, trimestre text not null, inicio date, fin date, generado text, primary key (ficha, trimestre) ); -- Cada bloque de clase del horario: día, horas, aula, materia, instructor y resultados create table if not exists horario_bloques ( ficha text not null, trimestre text not null, dia smallint not null check (dia between 0 and 5), -- 0 = lunes ... 5 = sábado hora_ini text not null, hora_fin text not null, horas numeric, aula text, materia text, raps jsonb not null default '[]'::jsonb, -- [{codigo, texto}] instructor text, vinculacion text, primary key (ficha, trimestre, dia, hora_ini), foreign key (ficha, trimestre) references trimestres(ficha, trimestre) on delete cascade ); -- 2) SEGURIDAD: solo el instructor dueño de la ficha lee y escribe ---------- alter table trimestres enable row level security; alter table horario_bloques enable row level security; drop policy if exists trimestres_owner on trimestres; drop policy if exists horario_owner on horario_bloques; create policy trimestres_owner on trimestres for all to authenticated using (exists (select 1 from fichas f where f.numero = trimestres.ficha and f.owner = auth.uid())) with check (exists (select 1 from fichas f where f.numero = trimestres.ficha and f.owner = auth.uid())); create policy horario_owner on horario_bloques for all to authenticated using (exists (select 1 from fichas f where f.numero = horario_bloques.ficha and f.owner = auth.uid())) with check (exists (select 1 from fichas f where f.numero = horario_bloques.ficha and f.owner = auth.uid())); -- 3) ACCESO DEL APRENDIZ: ahora también devuelve el horario de su ficha ------ create or replace function aprendiz_login(p_doc text, p_ficha text, p_pin text) returns jsonb language plpgsql security definer set search_path = public as $$ declare r text; a aprendices; f fichas; j jsonb; asis jsonb; exc jsonb; tri jsonb; hor jsonb; begin r := _verificar_aprendiz(p_doc, p_ficha, p_pin); if r in ('bloqueado','pendiente','sin_pin') then return jsonb_build_object('error', r); end if; if r <> 'ok' then return null; end if; select * into a from aprendices where doc = trim(p_doc) and ficha = trim(p_ficha); select * into f from fichas where numero = a.ficha; select coalesce(jsonb_object_agg(resultado, jsonb_build_array(juicio, fecha, funcionario)), '{}'::jsonb) into j from juicios where ficha = a.ficha and doc = a.doc; select coalesce(jsonb_agg(jsonb_build_object( 'fecha', x.fecha, 'estado', x.estado, 'tema', (select s.tema from sesiones s where s.ficha = x.ficha and s.fecha = x.fecha) ) order by x.fecha), '[]'::jsonb) into asis from asistencia x where x.ficha = a.ficha and x.doc = a.doc; select coalesce(jsonb_agg(jsonb_build_object( 'fecha', e.fecha, 'estado', e.estado, 'comentario', e.comentario, 'nombre', e.nombre_archivo ) order by e.fecha), '[]'::jsonb) into exc from excusas e where e.ficha = a.ficha and e.doc = a.doc; select coalesce(jsonb_agg(jsonb_build_object( 'trimestre', t.trimestre, 'inicio', t.inicio, 'fin', t.fin ) order by t.trimestre), '[]'::jsonb) into tri from trimestres t where t.ficha = a.ficha; select coalesce(jsonb_agg(jsonb_build_object( 'trimestre', b.trimestre, 'dia', b.dia, 'ini', b.hora_ini, 'fin', b.hora_fin, 'horas', b.horas, 'aula', b.aula, 'materia', b.materia, 'raps', b.raps, 'instructor', b.instructor, 'vinculacion', b.vinculacion ) order by b.trimestre, b.dia, b.hora_ini), '[]'::jsonb) into hor from horario_bloques b where b.ficha = a.ficha; return jsonb_build_object( 'ficha', to_jsonb(f) - 'owner', 'aprendiz', to_jsonb(a) - 'pin_hash', 'juicios', j, 'asistencia', asis, 'excusas', exc, 'trimestres', tri, 'horario', hor ); end; $$; revoke all on function aprendiz_login(text, text, text) from public; grant execute on function aprendiz_login(text, text, text) to anon, authenticated;
